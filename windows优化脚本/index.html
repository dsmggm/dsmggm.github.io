@echo off
%1 mshta vbscript:CreateObject("Shell.Application").ShellExecute("cmd.exe","/c %~s0 ::","","runas",1)(window.close)&&exit
cd /d "%~dp0"
title windows辅助工具

:menu
cls
echo windows辅助工具
echo 版本:20250425
echo 作者:dsmggm
echo 说明:仅限win10/11使用
echo -----------------
echo 请选择操作：
echo -----------------
echo 1. 执行2-6操作
echo 2. 从不合并任务栏
echo 3. 资源管理器默认打开此电脑
echo 4. 显示文件拓展名
echo 5. 显示秒数
echo 6. 启用卓越性能
echo 7. windows11经典右键菜单
echo a. 禁用更新(待测试)
echo -----------------
echo q. 退出 
echo -----------------
set /p choice=请输入选择：

if "%choice%"=="1" goto 1
if "%choice%"=="2" goto 2
if "%choice%"=="3" goto 3
if "%choice%"=="4" goto 4
if "%choice%"=="5" goto 5
if "%choice%"=="6" goto 6
if "%choice%"=="7" goto 7
if "%choice%"=="a" goto a

if "%choice%"=="q" goto quit


echo 输入错误，按任意键返回菜单....
pause >nul
goto menu

:1
echo 2. 从不合并任务栏
reg add "HKEY_CURRENT_USER\Software\Microsoft\Windows\CurrentVersion\Explorer\Advanced" /v TaskbarGlomLevel /t REG_DWORD /d 2 /f

echo 3. 资源管理器默认打开此电脑
reg add "HKEY_CURRENT_USER\Software\Microsoft\Windows\CurrentVersion\Explorer\Advanced" /v LaunchTo /t REG_DWORD /d 1 /f

echo 4. 显示文件拓展名
reg add "HKEY_CURRENT_USER\Software\Microsoft\Windows\CurrentVersion\Explorer\Advanced" /v HideFileExt /t REG_DWORD /d 0 /f

echo 5. 显示秒数
reg add "HKEY_CURRENT_USER\Software\Microsoft\Windows\CurrentVersion\Explorer\Advanced" /v ShowSecondsInSystemClock /t REG_DWORD /d 1 /f

echo 6. 启用卓越性能
powercfg -duplicatescheme e9a42b02-d5df-448d-aa00-03f14749eb61

echo 7. 启用windows经典右键菜单
reg.exe add "HKCU\Software\Classes\CLSID\{86ca1aa0-34aa-4e8b-a509-50c905bae2a2}\InprocServer32" /f /ve
taskkill /f /im explorer.exe & start explorer.exe

echo 按任意键返回菜单....
pause >nul
goto menu

:2
echo 从不合并任务栏
reg add "HKEY_CURRENT_USER\Software\Microsoft\Windows\CurrentVersion\Explorer\Advanced" /v TaskbarGlomLevel /t REG_DWORD /d 2 /f
echo 按任意键返回菜单....
pause >nul
goto menu

:3
echo 资源管理器默认打开此电脑
reg add "HKEY_CURRENT_USER\Software\Microsoft\Windows\CurrentVersion\Explorer\Advanced" /v LaunchTo /t REG_DWORD /d 1 /f
echo 按任意键返回菜单....
pause >nul
goto menu

:4
echo 显示文件拓展名
reg add "HKEY_CURRENT_USER\Software\Microsoft\Windows\CurrentVersion\Explorer\Advanced" /v HideFileExt /t REG_DWORD /d 0 /f
echo 按任意键返回菜单....
pause >nul
goto menu

:5
echo 显示秒数
reg add "HKEY_CURRENT_USER\Software\Microsoft\Windows\CurrentVersion\Explorer\Advanced" /v ShowSecondsInSystemClock /t REG_DWORD /d 1 /f
echo 按任意键返回菜单....
pause >nul
goto menu

:6
echo 启用卓越性能
powercfg -duplicatescheme e9a42b02-d5df-448d-aa00-03f14749eb61
echo 按任意键返回菜单....
pause >nul
goto menu

:7
echo 启用windows经典右键菜单
reg.exe add "HKCU\Software\Classes\CLSID\{86ca1aa0-34aa-4e8b-a509-50c905bae2a2}\InprocServer32" /f /ve
taskkill /f /im explorer.exe & start explorer.exe
echo 按任意键返回菜单....
pause >nul
goto menu

:a
echo 禁用更新
sc stop wuauserv
sc config wuauserv start= disabled
sc stop UsoSvc
sc config UsoSvc start= disabled
reg add "HKLM\SOFTWARE\Policies\Microsoft\Windows\WindowsUpdate" /v AUOptions /t REG_DWORD /d 1 /f
reg add "HKLM\SOFTWARE\Policies\Microsoft\Windows\WindowsUpdate" /v IncludeRecommendedUpdates /t REG_DWORD /d 0 /f

:quit
exit
