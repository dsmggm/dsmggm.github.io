linux备份脚本

#!/bin/bash # 遇到错误时停止脚本 set -e : ' 这是块注释内容 # 仅打包不压缩 tar -cvf 备份文件名.tar 源文件或目录 # 使用gzip压缩（.tar.gz或.tgz） tar -zcvf 备份文件名.tar.gz 源文件或目录 # 使用bzip2压缩（.tar.bz2） tar -jcvf 备份文件名.tar.bz2 源文件或目录 # 使用xz压缩（.tar.xz，压缩率最高） tar -Jcvf 备份文件名.tar.xz 源文件或目录 -c：创建新的归档文件 -v：显示详细的打包过程 -f：指定备份文件的名称 -z：使用 gzip 压缩 -j：使用 bzip2 压缩 -J：使用 xz 压缩 -x：解压文件 -t：查看归档文件内容 时间变量 年 月 日 时 分 秒 %Y %m %d %H %M %S echo $(date +%Y%m%d%H%M%S) ' # 时间变量 TIMESTAMP=$(date +%Y%m%d%H%M%S) # 定义备份源 SOURCE_DIR="/root/" # 定义存放文件 BACKUP_DIR="/root/backup_$TIMESTAMP.tar.xz" # 创建备份目录 mkdir -p "$(dirname $BACKUP_DIR)" # 或使用 install -D /dev/null "$BACKUP_DIR" # 创建备份文件 tar -Jcvf "$BACKUP_DIR" "$SOURCE_DIR" echo "备份完成: $BACKUP_DIR"
